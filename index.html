import os import requests # Пары: "английское_слово": "русский_перевод" vocabulary = { "apple": "яблоко", "banana": "банан", "success": "успех", "developer": "разработчик", "journey": "путешествие", "knowledge": "знание", "confidence": "уверенность", "achieve": "достигать" } output_dir = "ru_audio" os.makedirs(output_dir, exist_ok=True) print(f"Скачиваю русскую озвучку для {len(vocabulary)} слов...") for en_word, ru_word in vocabulary.items(): # Параметр tl=ru включает русскую озвучку url = f"https://translate.google.com/translate_tts?ie=UTF-8&client=gtx&tl=ru&q={ru_word}" headers = {"User-Agent": "Mozilla/5.0"} try: response = requests.get(url, headers=headers) if response.status_code == 200: # Сохраняем под английским именем для надёжных URL-ссылок file_path = os.path.join(output_dir, f"{en_word}.mp3") with open(file_path, "wb") as f: f.write(response.content) print(f"✅ Скачано: {en_word}.mp3 ({ru_word})") else: print(f"❌ Ошибка для '{ru_word}' (Код: {response.status_code})") except Exception as e: print(f"❌ Ошибка соединения: {e}") print("\nГотово! Файлы сохранены в папку:", os.path.abspath(output_dir))
